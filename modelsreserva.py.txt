from datetime import datetime, timedelta

from exceptions.excepciones import (
    ReservaInvalidaError,
    ReservaNoEncontradaError,
    ReservaEstadoError,
    ConflictoHorarioError,
    ClienteNoEncontradoError,
    ServicioNoDisponibleError,
    CalculoCostoError
)

from utils.logger import (
    registrar_evento,
    registrar_error,
    registrar_advertencia
)


class Reserva:
    """Representa una reserva realizada por un cliente."""

    ESTADOS_VALIDOS = (
        "Pendiente",
        "Confirmada",
        "Cancelada",
        "Procesada"
    )

    def __init__(
        self,
        codigo,
        cliente,
        servicio,
        cantidad,
        fecha_inicio,
        duracion_horas
    ):
        try:
            if not isinstance(codigo, str) or not codigo.strip():
                raise ReservaInvalidaError(
                    "El código de reserva es obligatorio."
                )

            if cliente is None:
                raise ClienteNoEncontradoError(
                    "Debe asociar un cliente a la reserva."
                )

            if servicio is None:
                raise ServicioNoDisponibleError(
                    "Debe seleccionar un servicio."
                )

            if (
                not isinstance(cantidad, (int, float))
                or isinstance(cantidad, bool)
            ):
                raise ReservaInvalidaError(
                    "La cantidad debe ser numérica."
                )

            if cantidad <= 0:
                raise ReservaInvalidaError(
                    "La cantidad debe ser mayor que cero."
                )

            if not isinstance(fecha_inicio, datetime):
                raise ReservaInvalidaError(
                    "La fecha de inicio debe ser un objeto datetime."
                )

            if (
                not isinstance(duracion_horas, (int, float))
                or isinstance(duracion_horas, bool)
                or duracion_horas <= 0
            ):
                raise ReservaInvalidaError(
                    "La duración en horas debe ser mayor que cero."
                )

            self.__codigo = codigo.strip()
            self.__cliente = cliente
            self.__servicio = servicio
            self.__cantidad = float(cantidad)
            self.__fecha_inicio = fecha_inicio
            self.__duracion_horas = float(duracion_horas)

            self.__fecha_fin = fecha_inicio + timedelta(
                hours=duracion_horas
            )

            self.__estado = "Pendiente"
            self.__costo_total = 0.0

            registrar_evento(
                f"Reserva creada: {self.__codigo}"
            )

        except ReservaInvalidaError as error:
            registrar_error(error, "Creación de reserva")
            raise

        except Exception as error:
            registrar_error(
                error,
                "Error inesperado al crear reserva"
            )
            raise ReservaInvalidaError(
                "No fue posible crear la reserva."
            ) from error

    @property
    def codigo(self):
        return self.__codigo

    @property
    def cliente(self):
        return self.__cliente

    @property
    def servicio(self):
        return self.__servicio

    @property
    def estado(self):
        return self.__estado

    @property
    def costo_total(self):
        return self.__costo_total

    @property
    def fecha_inicio(self):
        return self.__fecha_inicio

    @property
    def fecha_fin(self):
        return self.__fecha_fin

    def calcular_costo(self, impuesto=0.19, descuento=0.0):
        """Calcula el costo aplicando impuesto y descuento."""

        try:
            if not 0 <= impuesto <= 1:
                raise CalculoCostoError(
                    "El impuesto debe estar entre 0 y 1."
                )

            if not 0 <= descuento <= 1:
                raise CalculoCostoError(
                    "El descuento debe estar entre 0 y 1."
                )

            subtotal = self.__servicio.calcular_costo(
                self.__cantidad
            )

            if not isinstance(subtotal, (int, float)):
                raise CalculoCostoError(
                    "El servicio devolvió un costo inválido."
                )

            if subtotal < 0:
                raise CalculoCostoError(
                    "El costo no puede ser negativo."
                )

            valor_descuento = subtotal * descuento
            subtotal_descuento = subtotal - valor_descuento
            valor_impuesto = subtotal_descuento * impuesto

            self.__costo_total = round(
                subtotal_descuento + valor_impuesto,
                2
            )

            registrar_evento(
                f"Costo calculado para reserva {self.__codigo}: "
                f"${self.__costo_total:.2f}"
            )

            return self.__costo_total

        except CalculoCostoError as error:
            registrar_error(error, "Cálculo de costo")
            raise

        except Exception as error:
            registrar_error(
                error,
                "Cálculo de costo de reserva"
            )
            raise CalculoCostoError(
                "No fue posible calcular el costo."
            ) from error

    def confirmar(self):
        """Confirma una reserva pendiente."""

        try:
            if self.__estado != "Pendiente":
                raise ReservaEstadoError(
                    "Solo se pueden confirmar reservas pendientes."
                )

            self.__estado = "Confirmada"

            registrar_evento(
                f"Reserva confirmada: {self.__codigo}"
            )

            return True

        except ReservaEstadoError as error:
            registrar_error(error, "Confirmación")
            raise

    def cancelar(self):
        """Cancela una reserva pendiente o confirmada."""

        try:
            if self.__estado not in ("Pendiente", "Confirmada"):
                raise ReservaEstadoError(
                    "La reserva no puede cancelarse en su estado actual."
                )

            self.__estado = "Cancelada"

            registrar_advertencia(
                f"Reserva cancelada: {self.__codigo}"
            )

            return True

        except ReservaEstadoError as error:
            registrar_error(error, "Cancelación")
            raise

    def procesar(self):
        """Procesa una reserva confirmada."""

        try:
            if self.__estado != "Confirmada":
                raise ReservaEstadoError(
                    "Solo se pueden procesar reservas confirmadas."
                )

            if self.__costo_total <= 0:
                self.calcular_costo()

            self.__estado = "Procesada"

            registrar_evento(
                f"Reserva procesada: {self.__codigo}"
            )

            return self.__costo_total

        except ReservaEstadoError as error:
            registrar_error(error, "Procesamiento")
            raise


class GestorReservas:
    """Administra las reservas mediante una lista en memoria."""

    def __init__(self):
        self.__reservas = []

    @property
    def reservas(self):
        return tuple(self.__reservas)

    def crear_reserva(
        self,
        codigo,
        cliente,
        servicio,
        cantidad,
        fecha_inicio,
        duracion_horas
    ):
        """Crea una reserva y verifica conflictos horarios."""

        try:
            if any(
                reserva.codigo == codigo
                for reserva in self.__reservas
            ):
                raise ReservaInvalidaError(
                    "Ya existe una reserva con ese código."
                )

            nueva = Reserva(
                codigo,
                cliente,
                servicio,
                cantidad,
                fecha_inicio,
                duracion_horas
            )

            self.validar_disponibilidad(nueva)

            self.__reservas.append(nueva)

            registrar_evento(
                f"Reserva agregada al gestor: {codigo}"
            )

            return nueva

        except (
            ReservaInvalidaError,
            ConflictoHorarioError
        ) as error:
            registrar_error(error, "Gestor de reservas")
            raise

        except Exception as error:
            registrar_error(error, "Creación en gestor")
            raise ReservaInvalidaError(
                "No se pudo registrar la reserva."
            ) from error

    def validar_disponibilidad(self, nueva):
        """Evita cruces horarios para un mismo servicio."""

        for existente in self.__reservas:

            if existente.estado not in (
                "Pendiente",
                "Confirmada"
            ):
                continue

            mismo_servicio = (
                existente.servicio.obtener_codigo()
                == nueva.servicio.obtener_codigo()
            )

            hay_cruce = (
                nueva.fecha_inicio < existente.fecha_fin
                and nueva.fecha_fin > existente.fecha_inicio
            )

            if mismo_servicio and hay_cruce:
                raise ConflictoHorarioError(
                    "El servicio ya tiene una reserva "
                    "en ese horario."
                )

        return True

    def buscar_reserva(self, codigo):
        """Busca una reserva por su código."""

        for reserva in self.__reservas:
            if reserva.codigo == codigo:
                return reserva

        error = ReservaNoEncontradaError(
            f"No existe la reserva {codigo}."
        )

        registrar_error(error, "Búsqueda de reserva")
        raise error

    def confirmar_reserva(self, codigo):
        return self.buscar_reserva(codigo).confirmar()

    def cancelar_reserva(self, codigo):
        return self.buscar_reserva(codigo).cancelar()

    def procesar_reserva(self, codigo):
        return self.buscar_reserva(codigo).procesar()

    def listar_reservas(self):
        """Devuelve información de todas las reservas."""

        return [
            {
                "codigo": reserva.codigo,
                "cliente": reserva.cliente.nombre,
                "servicio": reserva.servicio.obtener_codigo(),
                "estado": reserva.estado,
                "costo": reserva.costo_total,
                "inicio": reserva.fecha_inicio
            }
            for reserva in self.__reservas
        ]
